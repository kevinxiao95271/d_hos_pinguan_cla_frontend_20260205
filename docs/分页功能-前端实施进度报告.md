## 🛠️ 快速实施模板

### 单列表页面模板

```vue
<script setup>
import { usePagination } from '@/composables/usePagination'

// 1. 初始化分页
const {
  currentPage,
  pageSize,
  totalCount,
  pageSizes,
  showPagination,
  extractDataList,
  resetPagination,
  getPaginationParams
} = usePagination({ defaultPageSize: 50 })

// 2. 修改数据加载函数
const loadData = async () => {
  loading.value = true
  try {
    const res = await someApi({
      ...filters,
      ...getPaginationParams() // 添加分页参数
    })
    if (res.success) {
      dataList.value = extractDataList(res.data) // 使用 extractDataList
    }
  } catch (error) {
    console.error('加载失败:', error)
  } finally {
    loading.value = false
  }
}

// 3. 筛选时重置分页
const handleFilter = () => {
  resetPagination()
  loadData()
}
</script>

<template>
  <div>
    <!-- 原有表格 -->
    <el-table :data="dataList" v-loading="loading">
      <!-- ... -->
    </el-table>

    <!-- 添加分页器 -->
    <div v-if="showPagination" class="pagination-container">
      <el-pagination
        v-model:current-page="currentPage"
        v-model:page-size="pageSize"
        :total="totalCount"
        :page-sizes="pageSizes"
        layout="total, sizes, prev, pager, next, jumper"
        @size-change="loadData"
        @current-change="loadData"
      />
    </div>
  </div>
</template>

<style scoped>
.pagination-container {
  margin-top: 20px;
  display: flex;
  justify-content: flex-end;
}
</style>
```
